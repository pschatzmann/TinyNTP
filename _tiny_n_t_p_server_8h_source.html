60 bool process() {
61 if (_source == nullptr) return false;
62 int size = _udp.parsePacket();
63 if (size < PacketSize) return false;
64
65 // the time when the request was received
66 uint64_t rxMs = _source->getTimeMs();
67 uint8_t request[PacketSize];
68 int total = 0;
69 while (total < PacketSize) {
70 int len = _udp.read(request + total, PacketSize - total);
71 if (len <= 0) break;
72 total += len;
73 }
74 if (total < PacketSize) return false;
75 if ((request[0] & 0x07) != 3) return false; // only client requests
76 if (rxMs == 0) return false; // time not known yet
77
78 uint8_t version = (request[0] >> 3) & 0x07;
79 if (version < 1 || version > 4) version = 4;
80
81 uint8_t response[PacketSize] = {};
82 response[0] = (version << 3) | 4; // LI=0, VN=version, Mode=4 (server)
83 response[1] = _stratum;
84 response[2] = request[2]; // poll
85 response[3] = static_cast<uint8_t>(-10); // precision: 2^-10 s (~1 ms)
86 putUint32(response + 8, 0x00000400); // root dispersion: ~15 ms
87 putTimestamp(response + 16, rxMs); // reference time
88 for (int i = 0; i < 8; i++) {
89 response[24 + i] = request[40 + i]; // originate = client transmit time
90 }
91 putTimestamp(response + 32, rxMs); // receive time
92 putTimestamp(response + 40, _source->getTimeMs()); // transmit time
93
94 auto ip = _udp.remoteIP();
95 auto port = _udp.remotePort();
96 _udp.beginPacket(ip, port);
97 _udp.write(response, PacketSize);
98 _udp.endPacket();
99 _requests++;
100 return true;
101 }
