66 bool process() {
67 if (_source == nullptr) return false;
68 int size = _udp.parsePacket();
69 if (size <= 0) return false;
70
71 // the time when the request was received
72 uint64_t rxMs = _source->getTimeMs();
73 uint8_t request[MaxRequestSize];
74 int total = 0;
75 while (total < size && total < MaxRequestSize) {
76 int len = _udp.read(request + total, MaxRequestSize - total);
77 if (len <= 0) break;
78 total += len;
79 }
80 // mode 7: request to set the time
81 if (TinyTimeSetMessage::isPrivate(request, total)) {
82 return processTimeSet(request, total);
83 }
84 if (total < PacketSize) return false;
85 if ((request[0] & 0x07) != 3) return false; // only client requests
86 if (rxMs == 0) return false; // time not known yet
87
88 uint8_t version = (request[0] >> 3) & 0x07;
89 if (version < 1 || version > 4) version = 4;
90
91 uint8_t response[PacketSize] = {};
92 response[0] = (version << 3) | 4; // LI=0, VN=version, Mode=4 (server)
93 response[1] = _stratum;
94 response[2] = request[2]; // poll
95 response[3] = static_cast<uint8_t>(-10); // precision: 2^-10 s (~1 ms)
96 putUint32(response + 8, 0x00000400); // root dispersion: ~15 ms
97 putTimestamp(response + 16, rxMs); // reference time
98 for (int i = 0; i < 8; i++) {
99 response[24 + i] = request[40 + i]; // originate = client transmit time
100 }
101 putTimestamp(response + 32, rxMs); // receive time
102 putTimestamp(response + 40, _source->getTimeMs()); // transmit time
103
104 auto ip = _udp.remoteIP();
105 auto port = _udp.remotePort();
106 _udp.beginPacket(ip, port);
107 _udp.write(response, PacketSize);
108 _udp.endPacket();
109 _requests++;
110 return true;
111 }

197 bool processTimeSet(const uint8_t* request, int size) {
198 if (_secretLen == 0) return false; // not allowed
199 uint8_t response[TinyTimeSetMessage::SetRequestSize];
200 if (TinyTimeSetMessage::type(request) ==
201 TinyTimeSetMessage::ChallengeRequest) {
202 // a nonce that nobody can predict without the secret
203 uint32_t now = ::millis();
204 TinyTimeSetMessage::makeNonce(_secret, _secretLen, ++_nonceCounter,
205 _source->getTimeMs(), now, _nonce);
206 _nonceValid = true;
207 _nonceMillis = now;
208 reply(response,
209 TinyTimeSetMessage::makeChallengeResponse(response, _nonce));
210 return true;
211 }
212 if (TinyTimeSetMessage::type(request) != TinyTimeSetMessage::SetRequest) {
213 return false;
214 }
215 TinyTimeSetMessage::Status status = TinyTimeSetMessage::Rejected;
216 uint64_t timeMs = 0;
217 // 32 bit arithmetic: also correct after the overflow of millis(), and
218 // if millis() returns a 64 bit value (e.g. on a computer)
219 bool fresh = _nonceValid && static_cast<uint32_t>(::millis() - _nonceMillis) <
220 NonceLifetimeMs;
221 // a failed attempt keeps the nonce: others cannot block a valid request
222 if (fresh && TinyTimeSetMessage::verifySetRequest(
223 request, size, _secret, _secretLen, _nonce, timeMs)) {
224 _nonceValid = false; // use it only once
225 if (timeMs < TinyTimeSetMessage::MinValidTimeMs) {
226 status = TinyTimeSetMessage::InvalidTime;
227 } else if (!_source->setTimeMs(timeMs)) {
228 status = TinyTimeSetMessage::NotSupported;
229 } else {
230 status = TinyTimeSetMessage::Ok;
231 _timeSetCount++;
232 }
233 }
234 reply(response, TinyTimeSetMessage::makeSetResponse(
235 response, status, _source->getTimeMs()));
236 return true;
237 }

248 static void putTimestamp(uint8_t* out, uint64_t unixMs) {
249 uint32_t seconds = static_cast<uint32_t>(unixMs / 1000) + NtpEpochOffset;
250 uint32_t fraction = static_cast<uint32_t>(
251 (static_cast<uint64_t>(unixMs % 1000) << 32) / 1000);
252 putUint32(out, seconds);
253 putUint32(out + 4, fraction);
254 }
