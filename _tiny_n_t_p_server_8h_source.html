62 bool process() {
63 if (_getTimeMs == nullptr) return false;
64 int size = _udp.parsePacket();
65 if (size < PacketSize) return false;
66
67 // the time when the request was received
68 uint64_t rxMs = _getTimeMs(_source);
69 uint8_t request[PacketSize];
70 int total = 0;
71 while (total < PacketSize) {
72 int len = _udp.read(request + total, PacketSize - total);
73 if (len <= 0) break;
74 total += len;
75 }
76 if (total < PacketSize) return false;
77 if ((request[0] & 0x07) != 3) return false; // only client requests
78 if (rxMs == 0) return false; // time not known yet
79
80 uint8_t version = (request[0] >> 3) & 0x07;
81 if (version < 1 || version > 4) version = 4;
82
83 uint8_t response[PacketSize] = {};
84 response[0] = (version << 3) | 4; // LI=0, VN=version, Mode=4 (server)
85 response[1] = _stratum;
86 response[2] = request[2]; // poll
87 response[3] = static_cast<uint8_t>(-10); // precision: 2^-10 s (~1 ms)
88 putUint32(response + 8, 0x00000400); // root dispersion: ~15 ms
89 putTimestamp(response + 16, rxMs); // reference time
90 for (int i = 0; i < 8; i++) {
91 response[24 + i] = request[40 + i]; // originate = client transmit time
92 }
93 putTimestamp(response + 32, rxMs); // receive time
94 putTimestamp(response + 40, _getTimeMs(_source)); // transmit time
95
96 auto ip = _udp.remoteIP();
97 auto port = _udp.remotePort();
98 _udp.beginPacket(ip, port);
99 _udp.write(response, PacketSize);
100 _udp.endPacket();
101 _requests++;
102 return true;
103 }
