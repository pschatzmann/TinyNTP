76 bool process() {
77 if (_source == nullptr) return false;
78 int size = _udp.parsePacket();
79 if (size <= 0) return false;
80 if (!acceptClient()) {
81 _dropped++;
82 uint8_t rest[16];
83 while (_udp.available() > 0 && _udp.read(rest, sizeof(rest)) > 0) {
84 }
85 return false;
86 }
87
88 // the time when the request was received
89 uint64_t rxMs = _source->getTimeMs();
90 uint8_t request[MaxRequestSize];
91 int total = 0;
92 while (total < size && total < MaxRequestSize) {
93 int len = _udp.read(request + total, MaxRequestSize - total);
94 if (len <= 0) break;
95 total += len;
96 }
97 // mode 7: request to set the time
98 if (TinyTimeSetMessage::isPrivate(request, total)) {
99 return processTimeSet(request, total);
100 }
101 if (total < PacketSize) return false;
102 if ((request[0] & 0x07) != 3) return false; // only client requests
103 if (rxMs == 0) return false; // time not known yet
104
105 uint8_t version = (request[0] >> 3) & 0x07;
106 if (version < 1 || version > 4) version = 4;
107
108 uint8_t response[PacketSize] = {};
109 response[0] = (_leapIndicator << 6) | (version << 3) | 4; // Mode=4 (server)
110 response[1] = _stratum;
111 response[2] = request[2]; // poll
112 response[3] = static_cast<uint8_t>(_precision); // e.g. 2^-10 s (~1 ms)
113 putUint32(response + 4, toShortFormat(_rootDelayMs));
114 putUint32(response + 8, toShortFormat(_rootDispersionMs));
115 memcpy(response + 12, _referenceId, 4);
116 putTimestamp(response + 16, _referenceTimeMs != 0 ? _referenceTimeMs : rxMs);
117 for (int i = 0; i < 8; i++) {
118 response[24 + i] = request[40 + i]; // originate = client transmit time
119 }
120 putTimestamp(response + 32, rxMs); // receive time
121 putTimestamp(response + 40, _source->getTimeMs()); // transmit time
122
123 auto ip = _udp.remoteIP();
124 auto port = _udp.remotePort();
125 _udp.beginPacket(ip, port);
126 _udp.write(response, PacketSize);
127 _udp.endPacket();
128 _requests++;
129 return true;
130 }

189 void setAllowedSubnets(const char* list) {
190 _subnetCount = 0;
191 _allowAll = false;
192 const char* p = list;
193 while (p != nullptr && *p && _subnetCount < MaxSubnets) {
194 while (*p == ' ' || *p == ',') p++;
195 if (!*p) break;
196 unsigned a, b, c, d, bits = 32;
197 int consumed = 0;
198 if (sscanf(p, "%u.%u.%u.%u%n", &a, &b, &c, &d, &consumed) == 4 && a < 256 && b < 256 && c < 256 && d < 256) {
199 const char* q = p + consumed;
200 if (*q == '/') bits = static_cast<unsigned>(atoi(q + 1));
201 if (bits > 32) bits = 32;
202 if (bits == 0) _allowAll = true;
203 _subnets[_subnetCount].mask = bits == 0 ? 0 : 0xFFFFFFFFu << (32 - bits);
204 _subnets[_subnetCount].net = ((a << 24) | (b << 16) | (c << 8) | d) & _subnets[_subnetCount].mask;
205 _subnetCount++;
206 }
207 p = strchr(p, ',');
208 }
209 if (_subnetCount == 0) _allowAll = true; // nothing valid: do not lock everybody out
210 }

324 bool acceptClient() {
325 auto ip = _udp.remoteIP();
326 const uint32_t addr = (static_cast<uint32_t>(ip[0]) << 24) | (static_cast<uint32_t>(ip[1]) << 16) |
327 (static_cast<uint32_t>(ip[2]) << 8) | static_cast<uint32_t>(ip[3]);
328 if (!_allowAll) {
329 bool ok = false;
330 for (int i = 0; i < _subnetCount && !ok; i++) ok = (addr & _subnets[i].mask) == _subnets[i].net;
331 if (!ok) return false;
332 }
333 if (_rateLimit > 0) {
334 const uint32_t now = ::millis();
335 Window* w = nullptr;
336 Window* oldest = &_windows[0];
337 for (Window& x : _windows) {
338 if (x.count > 0 && x.addr == addr) w = &x;
339 if (oldest->count != 0 && (x.count == 0 || now - x.startMs > now - oldest->startMs)) oldest = &x;
340 }
341 if (w == nullptr) {
342 w = oldest; // forgets the least recent client
343 w->addr = addr;
344 w->startMs = now;
345 w->count = 0;
346 }
347 if (static_cast<uint32_t>(now - w->startMs) >= 60000) {
348 w->startMs = now;
349 w->count = 0;
350 }
351 if (++w->count > _rateLimit) return false;
352 }
353 snprintf(_lastClient, sizeof(_lastClient), "%u.%u.%u.%u", static_cast<unsigned>(ip[0]),
354 static_cast<unsigned>(ip[1]), static_cast<unsigned>(ip[2]), static_cast<unsigned>(ip[3]));
355 return true;
356 }

368 bool processTimeSet(const uint8_t* request, int size) {
369 if (_secretLen == 0) return false; // not allowed
370 uint8_t response[TinyTimeSetMessage::SetRequestSize];
371 if (TinyTimeSetMessage::type(request) ==
372 TinyTimeSetMessage::ChallengeRequest) {
373 // a nonce that nobody can predict without the secret
374 uint32_t now = ::millis();
375 TinyTimeSetMessage::makeNonce(_secret, _secretLen, ++_nonceCounter,
376 _source->getTimeMs(), now, _nonce);
377 _nonceValid = true;
378 _nonceMillis = now;
379 reply(response,
380 TinyTimeSetMessage::makeChallengeResponse(response, _nonce));
381 return true;
382 }
383 if (TinyTimeSetMessage::type(request) != TinyTimeSetMessage::SetRequest) {
384 return false;
385 }
386 TinyTimeSetMessage::Status status = TinyTimeSetMessage::Rejected;
387 uint64_t timeMs = 0;
388 // 32 bit arithmetic: also correct after the overflow of millis(), and
389 // if millis() returns a 64 bit value (e.g. on a computer)
390 bool fresh = _nonceValid && static_cast<uint32_t>(::millis() - _nonceMillis) <
391 NonceLifetimeMs;
392 // a failed attempt keeps the nonce: others cannot block a valid request
393 if (fresh && TinyTimeSetMessage::verifySetRequest(
394 request, size, _secret, _secretLen, _nonce, timeMs)) {
395 _nonceValid = false; // use it only once
396 if (timeMs < TinyTimeSetMessage::MinValidTimeMs) {
397 status = TinyTimeSetMessage::InvalidTime;
398 } else if (!_source->setTimeMs(timeMs)) {
399 status = TinyTimeSetMessage::NotSupported;
400 } else {
401 status = TinyTimeSetMessage::Ok;
402 _timeSetCount++;
403 }
404 }
405 reply(response, TinyTimeSetMessage::makeSetResponse(
406 response, status, _source->getTimeMs()));
407 return true;
408 }

419 static void putTimestamp(uint8_t* out, uint64_t unixMs) {
420 uint32_t seconds = static_cast<uint32_t>(unixMs / 1000) + NtpEpochOffset;
421 uint32_t fraction = static_cast<uint32_t>(
422 (static_cast<uint64_t>(unixMs % 1000) << 32) / 1000);
423 putUint32(out, seconds);
424 putUint32(out + 4, fraction);
