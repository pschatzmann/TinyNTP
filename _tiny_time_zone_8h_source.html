43 bool setTimeZone(const char* tz) {

95 bool isDst(int64_t utc) const {
96 if (!_hasDst) return false;
97 int year = yearOf(utc + _stdOffset);
98 int64_t start = transition(year, _start) - _stdOffset;
99 int64_t end = transition(year, _end) - _dstOffset;
100 if (start < end) return utc >= start && utc < end;
101 // southern hemisphere: DST spans the new year
102 return !(utc >= end && utc < start);
103 }

184 static bool parseName(const char*& p, char (&out)[8]) {
185 size_t n = 0;
186 if (*p == '<') {
187 p++;
188 while (*p && *p != '>') {
189 if (n < sizeof(out) - 1) out[n++] = *p;
190 p++;
191 }
192 if (*p != '>') return false;
193 p++;
194 } else {
195 while (isAlpha(*p)) {
196 if (n < sizeof(out) - 1) out[n++] = *p;
197 p++;
198 }
199 }
200 out[n] = 0;
201 return n >= 3 || (n > 0 && out[0] != 0 && p[-1] == '>');
202 }

213 static bool parseTime(const char*& p, int32_t& seconds) {
214 int sign = 1;
215 if (*p == '+' || *p == '-') sign = (*p++ == '-') ? -1 : 1;
216 int h = 0, m = 0, s = 0;
217 if (!parseInt(p, h)) return false;
218 if (*p == ':') {
219 p++;
220 if (!parseInt(p, m)) return false;
221 if (*p == ':') {
222 p++;
223 if (!parseInt(p, s)) return false;
224 }
225 }
226 seconds = sign * (h * 3600 + m * 60 + s);
227 return true;
228 }

231 static bool parseRule(const char*& p, Rule& r) {
232 if (*p == 'M') {
233 p++;
234 r.type = 'M';
235 if (!parseInt(p, r.month) || *p++ != '.' || !parseInt(p, r.week) ||
236 *p++ != '.' || !parseInt(p, r.day))
237 return false;
238 if (r.month < 1 || r.month > 12 || r.week < 1 || r.week > 5 ||
239 r.day > 6)
240 return false;
241 } else if (*p == 'J') {
242 p++;
243 r.type = 'J';
244 if (!parseInt(p, r.day) || r.day < 1 || r.day > 365) return false;
245 } else {
246 r.type = 'N';
247 if (!parseInt(p, r.day) || r.day > 365) return false;
248 }
249 r.time = 7200;
250 if (*p == '/') {
251 p++;
252 if (!parseTime(p, r.time)) return false;
253 }
254 return true;

269 static int64_t daysFromCivil(int y, int m, int d) {
270 y -= m <= 2;
271 int64_t era = (y >= 0 ? y : y - 399) / 400;
272 int64_t yoe = y - era * 400;
273 int64_t doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
274 int64_t doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
275 return era * 146097 + doe - 719468;
276 }

279 static int yearOf(int64_t t) {
280 int64_t days = t >= 0 ? t / 86400 : -((-t + 86399) / 86400);
281 int64_t z = days + 719468;
282 int64_t era = (z >= 0 ? z : z - 146096) / 146097;
283 int64_t doe = z - era * 146097;
284 int64_t yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
285 int64_t doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
286 int64_t mp = (5 * doy + 2) / 153;
287 int m = static_cast<int>(mp < 10 ? mp + 3 : mp - 9);
288 return static_cast<int>(yoe + era * 400 + (m <= 2));
289 }

292 static int64_t transition(int year, const Rule& r) {
293 int64_t days;
294 if (r.type == 'M') {
295 int64_t first = daysFromCivil(year, r.month, 1);
296 int wday = static_cast<int>(((first + 4) % 7 + 7) % 7); // 1970-01-01 = Thu
297 int day = 1 + (r.day - wday + 7) % 7 + (r.week - 1) * 7;
298 while (day > daysInMonth(year, r.month)) day -= 7;
299 days = first + day - 1;
300 } else if (r.type == 'J') {
301 // day 1-365, February 29 is never counted
302 days = daysFromCivil(year, 1, 1) + r.day - 1 +
303 ((isLeap(year) && r.day >= 60) ? 1 : 0);
304 } else {
305 days = daysFromCivil(year, 1, 1) + r.day;
306 }
307 return days * 86400 + r.time;
308 }
