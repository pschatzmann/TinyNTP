93 bool isDst(int64_t utc) const {
94 if (!_hasDst) return false;
95 int year = yearOf(utc + _stdOffset);
96 int64_t start = transition(year, _start) - _stdOffset;
97 int64_t end = transition(year, _end) - _dstOffset;
98 if (start < end) return utc >= start && utc < end;
99 // southern hemisphere: DST spans the new year
100 return !(utc >= end && utc < start);

167 size_t n = 0;
168 if (*p == '<') {

170 while (*p && *p != '>') {

194 static bool parseTime(const char*& p, int32_t& seconds) {
195 int sign = 1;
196 if (*p == '+' || *p == '-') sign = (*p++ == '-') ? -1 : 1;
197 int h = 0, m = 0, s = 0;
198 if (!parseInt(p, h)) return false;
199 if (*p == ':') {
200 p++;
201 if (!parseInt(p, m)) return false;
202 if (*p == ':') {
203 p++;
204 if (!parseInt(p, s)) return false;
205 }
206 }
207 seconds = sign * (h * 3600 + m * 60 + s);
208 return true;
209 }

212 static bool parseRule(const char*& p, Rule& r) {
213 if (*p == 'M') {
214 p++;
215 r.type = 'M';
216 if (!parseInt(p, r.month) || *p++ != '.' || !parseInt(p, r.week) ||
217 *p++ != '.' || !parseInt(p, r.day))
218 return false;
219 if (r.month < 1 || r.month > 12 || r.week < 1 || r.week > 5 ||
220 r.day > 6)
221 return false;
222 } else if (*p == 'J') {
223 p++;
224 r.type = 'J';
225 if (!parseInt(p, r.day) || r.day < 1 || r.day > 365) return false;
226 } else {
227 r.type = 'N';
228 if (!parseInt(p, r.day) || r.day > 365) return false;
229 }
230 r.time = 7200;
231 if (*p == '/') {
232 p++;
233 if (!parseTime(p, r.time)) return false;
234 }
235 return true;

248 static int64_t daysFromCivil(int y, int m, int d) {
249 y -= m <= 2;
250 int64_t era = (y >= 0 ? y : y - 399) / 400;
251 int64_t yoe = y - era * 400;
252 int64_t doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
253 int64_t doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
254 return era * 146097 + doe - 719468;
255 }

258 static int yearOf(int64_t t) {
259 int64_t days = t >= 0 ? t / 86400 : -((-t + 86399) / 86400);
260 int64_t z = days + 719468;
261 int64_t era = (z >= 0 ? z : z - 146096) / 146097;
262 int64_t doe = z - era * 146097;
263 int64_t yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
264 int64_t doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
265 int64_t mp = (5 * doy + 2) / 153;
266 int m = static_cast<int>(mp < 10 ? mp + 3 : mp - 9);
267 return static_cast<int>(yoe + era * 400 + (m <= 2));
268 }

271 static int64_t transition(int year, const Rule& r) {
272 int64_t days;
273 if (r.type == 'M') {
274 int64_t first = daysFromCivil(year, r.month, 1);
275 int wday = static_cast<int>(((first + 4) % 7 + 7) % 7); // 1970-01-01 = Thu
276 int day = 1 + (r.day - wday + 7) % 7 + (r.week - 1) * 7;
277 while (day > daysInMonth(year, r.month)) day -= 7;
278 days = first + day - 1;
279 } else if (r.type == 'J') {
280 // day 1-365, February 29 is never counted
281 days = daysFromCivil(year, 1, 1) + r.day - 1 +
282 ((isLeap(year) && r.day >= 60) ? 1 : 0);
283 } else {
284 days = daysFromCivil(year, 1, 1) + r.day;
285 }
286 return days * 86400 + r.time;
287 }
