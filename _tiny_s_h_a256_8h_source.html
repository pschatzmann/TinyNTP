38 void update(const uint8_t* data, size_t len) {
39 _bitCount += static_cast<uint64_t>(len) * 8;
40 while (len > 0) {
41 size_t n = BlockSize - _bufferLen;
42 if (n > len) n = len;
43 memcpy(_buffer + _bufferLen, data, n);
44 _bufferLen += n;
45 data += n;
46 len -= n;
47 if (_bufferLen == BlockSize) {
48 transform(_buffer);
49 _bufferLen = 0;
50 }
51 }
52 }

58 void finish(uint8_t* digest) {
59 uint64_t bits = _bitCount;
60 uint8_t pad = 0x80;
61 update(&pad, 1);
62 pad = 0;

66 update(length, 8);

69 digest[i * 4 + 1] = _state[i] >> 16;
70 digest[i * 4 + 2] = _state[i] >> 8;

155 static void calculate(const uint8_t* key, size_t keyLen, const uint8_t* data,
156 size_t dataLen, uint8_t* mac) {
157 uint8_t k[TinySHA256::BlockSize] = {};
158 if (keyLen > TinySHA256::BlockSize) {
159 TinySHA256 h;
160 h.update(key, keyLen);
161 h.finish(k);
162 } else {
163 memcpy(k, key, keyLen);
164 }
165 uint8_t pad[TinySHA256::BlockSize];
166 uint8_t inner[TinySHA256::DigestSize];
167 TinySHA256 h;
168 for (size_t i = 0; i < sizeof pad; i++) pad[i] = k[i] ^ 0x36;
169 h.update(pad, sizeof pad);
170 h.update(data, dataLen);
171 h.finish(inner);
172 h.reset();
173 for (size_t i = 0; i < sizeof pad; i++) pad[i] = k[i] ^ 0x5c;
174 h.update(pad, sizeof pad);
175 h.update(inner, sizeof inner);
176 h.finish(mac);
177 }
