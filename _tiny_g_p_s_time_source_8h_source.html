104 bool parseSentence() {
105 char* star = strchr(_buf, '*');
106 if (star == nullptr || hex(star[1]) < 0 || hex(star[2]) < 0) return false;
107 uint8_t sum = 0;
108 for (char* p = _buf; p < star; ++p) sum ^= static_cast<uint8_t>(*p);
109 if (sum != hex(star[1]) * 16 + hex(star[2])) {
110 _checksumErrors++;
111 return false;
112 }
113 *star = 0;
114 if (strlen(_buf) < 5) return false;
115 const char* type = _buf + 2; // after the talker
116 const char talker[3] = {_buf[0], _buf[1], 0};
117 char* f[24];
118 const int n = split(f, 24);
119 if (strncmp(type, "RMC", 3) == 0 && n >= 10) {
120 _rmcValid = f[2][0] == 'A';
121 return _rmcValid && setTime(f[1], f[9], nullptr, nullptr, nullptr);
122 }
123 if (strncmp(type, "ZDA", 3) == 0 && n >= 5) {
124 return f[2][0] && f[4][0] && setTime(f[1], nullptr, f[2], f[3], f[4]);
125 }
126 if (strncmp(type, "GGA", 3) == 0 && n >= 8) {
127 _fixQuality = atoi(f[6]);
128 _satsUsed = atoi(f[7]);
129 } else if (strncmp(type, "GSV", 3) == 0 && n >= 4) {
130 static const char* const talkers[6] = {"GP", "GL", "GA", "GB", "GQ", "GI"};
131 for (int i = 0; i < 6; ++i) {
132 if (strcmp(talker, talkers[i]) == 0) _satsInView[i] = atoi(f[3]);
133 }
134 }
135 return false;
136 }

141 bool setTime(const char* time, const char* ddmmyy, const char* day, const char* month,
142 const char* year) {
143 if (strlen(time) < 6) return false;
144 int d, m, y;
145 if (ddmmyy != nullptr) {
146 if (strlen(ddmmyy) < 6) return false;
147 d = two(ddmmyy);
148 m = two(ddmmyy + 2);
149 y = 2000 + two(ddmmyy + 4);
150 } else {
151 d = atoi(day);
152 m = atoi(month);
153 y = atoi(year);
154 }
155 if (m < 1 || m > 12 || d < 1 || d > 31 || y < 2000) return false;
156 const int hh = two(time), mm = two(time + 2), ss = two(time + 4);
157 if (hh > 23 || mm > 59 || ss > 60) return false;
158 _fractionMs = time[6] == '.' ? static_cast<int>(atof(time + 6) * 1000.0 + 0.5) : 0;
159 const int64_t days = daysFromCivil(y, m, d);
160 _timeMs = static_cast<uint64_t>(((days * 24 + hh) * 60 + mm) * 60 + ss) * 1000ULL + _fractionMs;
161 return true;
162 }

164 static int64_t daysFromCivil(int y, unsigned m, unsigned d) {
165 y -= m <= 2;
166 const int64_t era = (y >= 0 ? y : y - 399) / 400;
167 const unsigned yoe = static_cast<unsigned>(y - era * 400);
168 const unsigned doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
169 const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
170 return era * 146097 + static_cast<int64_t>(doe) - 719468;
171 }

227 bool feed(char c, uint32_t nowMicros) {
228 if (c == '$') _sentenceMicros = nowMicros;
229 if (!_parser.feed(c) || !_parser.hasFix()) return false;
230 const uint64_t utcMs = _parser.getTimeMs();
231 if (utcMs == _lastSentenceUtcMs) return false; // RMC and ZDA of the same second
232 _lastSentenceUtcMs = utcMs;
233 const uint32_t pps = _ppsMicros;
234 const uint32_t sincePps = _sentenceMicros - pps;
235 _lastFixPps = _ppsCount != 0 && _parser.getFractionMs() == 0 && sincePps < 1000000UL;
236 if (_lastFixPps) {
237 _fixMicros = pps; // the start of the reported second
238 _fixUtcMs = utcMs;
239 } else {
240 _fixMicros = _sentenceMicros;
241 _fixUtcMs = utcMs + _latencyMs;
242 }
243 _fixMillis = ::millis() - static_cast<uint32_t>((nowMicros - _fixMicros) / 1000);
244 _fixCount++;
245 return true;
246 }
