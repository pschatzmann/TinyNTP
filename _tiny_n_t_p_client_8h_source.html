442 bool readResponse(int packetSize, NTPPacket& response) {
443 uint8_t* buffer = reinterpret_cast<uint8_t*>(&response);
444 int totalRead = 0;
445 while (totalRead < (int)sizeof(NTPPacket) && _udp.available()) {
446 int bytesRead =
447 _udp.read(buffer + totalRead, (int)sizeof(NTPPacket) - totalRead);
448 if (bytesRead <= 0) break;
449 totalRead += bytesRead;
450 }
451 uint8_t rest[16];
452 while (_udp.available() > 0 && _udp.read(rest, sizeof(rest)) > 0) {
453 }
454 if (packetSize < (int)sizeof(NTPPacket)) {
455 log("NTP: packet too short - Expected: %d, Got: %d\n",
456 (int)sizeof(NTPPacket), packetSize);
457 return false;
458 }
459 if (totalRead < (int)sizeof(NTPPacket)) {
460 log("NTP: response read incomplete - Expected: %d, Got: %d\n",
461 (int)sizeof(NTPPacket), totalRead);
462 return false;
463 }
464 return true;
465 }

468 void storeResult(const NTPPacket& r, uint32_t roundTripMs) {
469 TinyNTPResult& res = _lastResult;
470 res.leapIndicator = r.li_vn_mode >> 6;
471 res.version = (r.li_vn_mode >> 3) & 0x07;
472 res.stratum = r.stratum;
473 res.precision = static_cast<int8_t>(r.precision);
474 res.rootDelayMs = l_ntohl(r.rootDelay) / 65536.0 * 1000.0;
475 res.rootDispersionMs = l_ntohl(r.rootDispersion) / 65536.0 * 1000.0;
476 res.referenceId = r.refId;
477 const uint8_t* id = reinterpret_cast<const uint8_t*>(&r.refId);
478 if (r.stratum <= 1) {
479 int n = 0;
480 for (int i = 0; i < 4; i++) {
481 if (id[i] >= 32 && id[i] < 127) res.referenceText[n++] = static_cast<char>(id[i]);
482 }
483 while (n > 0 && res.referenceText[n - 1] == ' ') n--;
484 res.referenceText[n] = 0;
485 } else {
486 snprintf(res.referenceText, sizeof(res.referenceText), "%u.%u.%u.%u", id[0], id[1], id[2], id[3]);
487 }
488 res.roundTripMs = roundTripMs;
489 res.delayMs = roundTripMs;
490 }

501 bool ntpExchange(uint32_t txTm_s) {
502 _lastResult.valid = false;
503 NTPPacket packet = {};
504 // Set LI=3 (clock not synchronized), VN=3 (NTPv3), Mode=3 (client)
505 // Binary: 11 011 011 = 0xDB
506 packet.li_vn_mode = 0xDB;
507 // Convert transmit timestamp to network byte order
508 packet.txTm_s = l_htonl(txTm_s);
509
510 // The socket stays open between the updates, so that a late response is
511 // still received (and ignored) and does not get lost with the socket
512 uint16_t localPort = _localPort != 0 ? _localPort : _port;
513 if (_openPort != localPort) {
514 _openPort = _udp.begin(localPort) ? localPort : 0;
515 }
516
517 // The request is sent again until a response arrives: some WiFi networks
518 // (e.g. with an ESP32) lose responses or hold them back until the device
519 // sends the next packet. Each request has its own number in the transmit
520 // fraction, which the server returns as originate time: the round trip is
521 // measured from the request that was answered, and responses to the
522 // requests of an earlier update are ignored.
523 uint32_t sentMillis[MaxRequests];
524 int sentCount = 0;
525 uint32_t requestId = ++_requestCounter << 8;
526 uint32_t startMillis = ::millis();
527 uint32_t t0_millis = 0; // the time when the answered request was sent
528 uint32_t t3_millis = 0; // the time when the response was received
529 NTPPacket response = {};
530 bool received = false;
531 while (!received) {
532 // 32 bit arithmetic: also correct after the overflow of millis(), and if
533 // millis() returns a 64 bit value (e.g. on a computer)
534 uint32_t now = ::millis();
535 if (static_cast<uint32_t>(now - startMillis) > _timeoutMs) {
536 log("NTP: request timed out\n");
537 return false;
538 }
539 if (sentCount < MaxRequests &&
540 (sentCount == 0 || static_cast<uint32_t>(
541 now - sentMillis[sentCount - 1]) >=
542 ResendIntervalMs)) {
543 packet.txTm_f = l_htonl(requestId | sentCount);
544 sentMillis[sentCount++] = now;
545 _udp.beginPacket(_server, _port);
546 _udp.write(reinterpret_cast<uint8_t*>(&packet), sizeof(NTPPacket));
547 _udp.endPacket();
548 }
549
550 int packetSize = _udp.parsePacket();
551 if (packetSize <= 0) continue;
552 t3_millis = ::millis();
553 if (!readResponse(packetSize, response)) continue;
554 uint32_t originate = l_ntohl(response.origTm_f);
555 int index = static_cast<int>(originate & 0xFF);
556 if ((originate & ~0xFFu) != requestId || index >= sentCount) continue;
557 t0_millis = sentMillis[index];
558 storeResult(response, t3_millis - t0_millis);
559 if (response.stratum == 0) {
560 log("NTP: kiss-o'-death %s\n", _lastResult.referenceText);
561 return false;
562 }
563 if (_rejectUnsynchronized &&
564 (_lastResult.leapIndicator == 3 || response.stratum >= 16)) {
565 log("NTP: server is not synchronised\n");
566 return false;
567 }
568 received = true;
569 }
570
571 uint32_t receive = l_ntohl(response.rxTm_s);
572 uint32_t receive_f = l_ntohl(response.rxTm_f);
573 uint32_t transmit = l_ntohl(response.txTm_s);
574 uint32_t transmit_f = l_ntohl(response.txTm_f);
575
576 // The transmit time of the server is the time when it sent the response.
577 // The response needed about half of the network delay to get to us, which
578 // is the time of the round trip without the time that the server needed.
579 uint32_t roundTripMs = t3_millis - t0_millis;
580 int64_t serverMs =
581 static_cast<int64_t>(static_cast<int32_t>(transmit - receive)) * 1000 +
582 static_cast<int64_t>(ntpFractionToMs(transmit_f)) -
583 static_cast<int64_t>(ntpFractionToMs(receive_f));
584 int64_t delayMs = static_cast<int64_t>(roundTripMs) - serverMs;
585 if (delayMs < 0) delayMs = 0;
586 if (delayMs > roundTripMs) delayMs = roundTripMs;
587 // Unix seconds, calculated with 32 bit (also if unsigned long has 64 bit):
588 // correct until 2106, also after the rollover of the NTP time in 2036
589 uint32_t transmitUnix = static_cast<uint32_t>(transmit - 2208988800UL);
590 uint64_t newTimeMs = static_cast<uint64_t>(transmitUnix) * 1000ULL +
591 ntpFractionToMs(transmit_f) +
592 static_cast<uint64_t>(delayMs / 2);
593 uint32_t newNtpTime = static_cast<uint32_t>(newTimeMs / 1000);
594 uint32_t newNtpTimeMs = static_cast<uint32_t>(newTimeMs % 1000);
595
596 if (!_hasBaseline) {
597 // First update: establish baseline for correction factor calculation
598 _baselineNtpTime = newNtpTime;
599 _baselineNtpTimeMs = newNtpTimeMs;
600 _baselineUpdateMillis = t3_millis;
601 _hasBaseline = true;
602 } else {
603 // Calculate correction factor based on cumulative drift since baseline
604 uint32_t raw_millis_elapsed = t3_millis - _baselineUpdateMillis;
605 // Calculate actual time elapsed according to NTP (with millisecond
606 // precision)
607 uint64_t actual_time_elapsed_ms =
608 ((uint64_t)(newNtpTime - _baselineNtpTime) * 1000ULL) +
609 newNtpTimeMs - _baselineNtpTimeMs;
610
611 // Only update correction factor if enough time has elapsed (at least 10
612 // seconds). This avoids division by zero and ensures meaningful
613 // calibration
614 if (raw_millis_elapsed >= 10000 && actual_time_elapsed_ms >= 10000) {
615 _millisCorrectionFactor = static_cast<double>(actual_time_elapsed_ms) /
616 static_cast<double>(raw_millis_elapsed);
617 }
618 }
619
620 _lastNtpTime = newNtpTime;
621 _lastNtpTimeMs = newNtpTimeMs;
622 _lastUpdateMillis = t3_millis;
623 _hasTime = true;
624 _lastResult.delayMs = static_cast<uint32_t>(delayMs);
625 _lastResult.serverTimeMs = newTimeMs;
626 _lastResult.receivedMillis = t3_millis;
627 _lastResult.valid = true;
628 return true;
629 }

636 bool updateWithoutRTC() {
637 // No real time clock: set transmit timestamp to zero, use millis for

647 bool updateWithRTC() {
648 // Use system clock for transmit timestamp (uncorrected to avoid circular dependency)
649 uint32_t t0_unix = getUncorrectedTimeSec(); // seconds since 1970
650 uint32_t t0_ntp = t0_unix + 2208988800UL; // seconds since 1900
651 return ntpExchange(t0_ntp);
