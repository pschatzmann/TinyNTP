77 uint64_t getTimeMs() {
78 if (_lastUpdateMillis == 0) {
79 return 0; // Time not yet initialized
80 }
81 // Calculate ms since last update with correction factor applied
82 uint32_t raw_ms_since_update = ::millis() - _lastUpdateMillis;

84 static_cast<double>(raw_ms_since_update) * _millisCorrectionFactor);
85 // Return last NTP time in ms plus corrected elapsed ms, plus offset
86 return (static_cast<uint64_t>(_lastNtpTime) * 1000ULL) +
87 _lastNtpTimeMs +

203 struct __attribute__((packed)) NTPPacket {
204 // Leap Indicator, Version Number, Mode (packed in one byte)
205 // Bit layout: LI(7-6) | VN(5-3) | Mode(2-0)
206 uint8_t li_vn_mode = 0; // Default: 0x00, set to 0xE3 for client request
207 uint8_t stratum = 0; // Stratum level of the local clock
208 uint8_t poll = 0; // Maximum interval between messages
209 uint8_t precision = 0; // Precision of the local clock
210 uint32_t rootDelay =
211 0; // Total round trip delay to the primary reference source
212 uint32_t rootDispersion =
213 0; // Max error relative to the primary reference source
214 uint32_t refId = 0; // Reference clock identifier
215 uint32_t refTm_s = 0; // Reference timestamp (seconds)
216 uint32_t refTm_f = 0; // Reference timestamp (fraction)
217 uint32_t origTm_s =
218 0; // Originate timestamp (seconds) - time request sent by client
219 uint32_t origTm_f = 0; // Originate timestamp (fraction)
220 uint32_t rxTm_s =
221 0; // Receive timestamp (seconds) - time request received by server
222 uint32_t rxTm_f = 0; // Receive timestamp (fraction)
223 uint32_t txTm_s =
224 0; // Transmit timestamp (seconds) - time reply sent by server
225 uint32_t txTm_f = 0; // Transmit timestamp (fraction)

308 bool ntpExchange(uint32_t txTm_s) {
309 NTPPacket packet = {};
310 // Set LI=3 (no warning), VN=3 (NTPv3), Mode=3 (client)
311 // Binary: 11 011 011 = 0xDB
312 packet.li_vn_mode = 0xDB;
313 // Convert transmit timestamp to network byte order
314 packet.txTm_s = l_htonl(txTm_s);
315 packet.txTm_f = 0;
316
317 _udp.begin(_localPort != 0 ? _localPort : _port);
318 _udp.beginPacket(_server, _port);
319 _udp.write(reinterpret_cast<uint8_t*>(&packet), sizeof(NTPPacket));
320 _udp.endPacket();
321
322 // Wait for response with timeout
323 uint32_t timeout_ms = ::millis() + _timeoutMs;
324 int packetSize = 0;
325 while ((packetSize = _udp.parsePacket()) == 0) {
326 if (::millis() > timeout_ms) {
327 log("NTP: request timed out\n");
328 return false;
329 }
330 }
331
332 // Read response
333 uint32_t t3_millis = ::millis();
334 NTPPacket response = {};
335 uint8_t* buffer = reinterpret_cast<uint8_t*>(&response);
336
337 // Check if packet size is correct
338 if (packetSize < (int)sizeof(NTPPacket)) {
339 _udp.read(buffer, packetSize); // Read available data for logging
340 log("NTP: packet too short - Expected: %d, Got: %d (%s)\n",
341 sizeof(NTPPacket), packetSize, buffer);
342 return false;
343 }
344
345 // Read the packet data - read in chunks if necessary
346 int totalRead = 0;
347 int bytesToRead = sizeof(NTPPacket);
348
349 while (totalRead < bytesToRead && _udp.available()) {
350 int bytesRead = _udp.read(buffer + totalRead, bytesToRead - totalRead);
351 if (bytesRead > 0) {
352 totalRead += bytesRead;
353 } else {
354 break;
355 }
356 }
357
358 if (totalRead < (int)sizeof(NTPPacket)) {
359 log("NTP: response read incomplete - Expected: %d, Got: %d (%s)\n",
360 sizeof(NTPPacket), totalRead, buffer);
361 return false;
362 }
363
364 bool useOffset = (txTm_s != 0);
365 if (useOffset) {
366 // Full NTP offset calculation
367 uint32_t originate = l_ntohl(response.origTm_s);
368 uint32_t receive = l_ntohl(response.rxTm_s);
369 uint32_t transmit = l_ntohl(response.txTm_s);
370 uint32_t transmit_f = l_ntohl(response.txTm_f);
371 // log("NTP: originate=%u, receive=%u, transmit=%u\n", originate, receive,
372 // transmit);
373 uint32_t t3_unix = getUncorrectedTimeSec();
374 uint32_t t3_ntp = t3_unix + 2208988800UL;
375 long offset =
376 ((long)(receive - originate) + (long)(transmit - t3_ntp)) / 2;
377 uint32_t newNtpTime = transmit - 2208988800UL + offset;
378 uint32_t newNtpTimeMs = ntpFractionToMs(transmit_f);
379
380 // Establish baseline on first update, calculate correction factor on subsequent updates
381 if (_baselineUpdateMillis == 0) {
382 // First update: establish baseline for correction factor calculation
383 _baselineNtpTime = newNtpTime;
384 _baselineNtpTimeMs = newNtpTimeMs;
385 _baselineUpdateMillis = t3_millis;
386 } else {
387 // Calculate correction factor based on cumulative drift since baseline
388 uint32_t raw_millis_elapsed = t3_millis - _baselineUpdateMillis;
389 // Calculate actual time elapsed according to NTP (with millisecond precision)
390 uint64_t actual_time_elapsed_ms =
391 ((uint64_t)(newNtpTime - _baselineNtpTime) * 1000ULL) +
392 newNtpTimeMs - _baselineNtpTimeMs;
393
394 // Only update correction factor if enough time has elapsed (at least 10 seconds)
395 // This avoids division by zero and ensures meaningful calibration
396 if (raw_millis_elapsed >= 10000 && actual_time_elapsed_ms >= 10000) {
397 _millisCorrectionFactor = static_cast<double>(actual_time_elapsed_ms) /
398 static_cast<double>(raw_millis_elapsed);
399 // log("NTP: correction factor = %.6f (actual=%llu ms, millis=%u ms from baseline)\n",
400 // _millisCorrectionFactor, actual_time_elapsed_ms, raw_millis_elapsed);
401 } else {
402 // log("NTP: skipping correction factor update - insufficient time from baseline (actual=%llu ms, millis=%u ms)\n",
403 // actual_time_elapsed_ms, raw_millis_elapsed);
404 }
405 }
406
407 _lastNtpTime = newNtpTime;
408 _lastNtpTimeMs = newNtpTimeMs;
409 _lastUpdateMillis = t3_millis;
410 } else {
411 // Only use server transmit timestamp (first update without RTC)
412 uint32_t transmit = l_ntohl(response.txTm_s);
413 uint32_t transmit_f = l_ntohl(response.txTm_f);
414 _lastNtpTime = transmit - 2208988800UL;
415 _lastNtpTimeMs = ntpFractionToMs(transmit_f);
416 _lastUpdateMillis = t3_millis;
417
418 // Establish baseline if not yet set
419 if (_baselineUpdateMillis == 0) {
420 _baselineNtpTime = _lastNtpTime;
421 _baselineNtpTimeMs = _lastNtpTimeMs;
422 _baselineUpdateMillis = t3_millis;
423 }
424 }
425 return true;
426 }

433 bool updateWithoutRTC() {
434 // No real time clock: set transmit timestamp to zero, use millis for
435 // timeout
436 return ntpExchange(0);
437 }
438
444 bool updateWithRTC() {
445 // Use system clock for transmit timestamp (uncorrected to avoid circular dependency)
446 uint32_t t0_unix = getUncorrectedTimeSec(); // seconds since 1970
447 uint32_t t0_ntp = t0_unix + 2208988800UL; // seconds since 1900
448 return ntpExchange(t0_ntp);
449 }
