351 bool ntpExchange(uint32_t txTm_s) {
352 NTPPacket packet = {};
353 // Set LI=3 (no warning), VN=3 (NTPv3), Mode=3 (client)
354 // Binary: 11 011 011 = 0xDB
355 packet.li_vn_mode = 0xDB;
356 // Convert transmit timestamp to network byte order
357 packet.txTm_s = l_htonl(txTm_s);
358 packet.txTm_f = 0;
359
360 _udp.begin(_localPort != 0 ? _localPort : _port);
361 uint32_t t0_millis = ::millis(); // the time when the request is sent
362 _udp.beginPacket(_server, _port);
363 _udp.write(reinterpret_cast<uint8_t*>(&packet), sizeof(NTPPacket));
364 _udp.endPacket();
365
366 // Wait for response with timeout
367 int packetSize = 0;
368 while ((packetSize = _udp.parsePacket()) == 0) {
369 // 32 bit arithmetic: also correct after the overflow of millis(), and if
370 // millis() returns a 64 bit value (e.g. on a computer)
371 if (static_cast<uint32_t>(::millis() - t0_millis) > _timeoutMs) {
372 log("NTP: request timed out\n");
373 return false;
374 }
375 }
376
377 // Read response
378 uint32_t t3_millis = ::millis();
379 NTPPacket response = {};
380 uint8_t* buffer = reinterpret_cast<uint8_t*>(&response);
381
382 // Check if packet size is correct
383 if (packetSize < (int)sizeof(NTPPacket)) {
384 _udp.read(buffer, packetSize); // Read available data for logging
385 log("NTP: packet too short - Expected: %d, Got: %d (%s)\n",
386 sizeof(NTPPacket), packetSize, buffer);
387 return false;
388 }
389
390 // Read the packet data - read in chunks if necessary
391 int totalRead = 0;
392 int bytesToRead = sizeof(NTPPacket);
393
394 while (totalRead < bytesToRead && _udp.available()) {
395 int bytesRead = _udp.read(buffer + totalRead, bytesToRead - totalRead);
396 if (bytesRead > 0) {
397 totalRead += bytesRead;
398 } else {
399 break;
400 }
401 }
402
403 if (totalRead < (int)sizeof(NTPPacket)) {
404 log("NTP: response read incomplete - Expected: %d, Got: %d (%s)\n",
405 sizeof(NTPPacket), totalRead, buffer);
406 return false;
407 }
408
409 uint32_t receive = l_ntohl(response.rxTm_s);
410 uint32_t receive_f = l_ntohl(response.rxTm_f);
411 uint32_t transmit = l_ntohl(response.txTm_s);
412 uint32_t transmit_f = l_ntohl(response.txTm_f);
413
414 // The transmit time of the server is the time when it sent the response.
415 // The response needed about half of the network delay to get to us, which
416 // is the time of the round trip without the time that the server needed.
417 uint32_t roundTripMs = t3_millis - t0_millis;
418 int64_t serverMs =
419 static_cast<int64_t>(static_cast<int32_t>(transmit - receive)) * 1000 +
420 static_cast<int64_t>(ntpFractionToMs(transmit_f)) -
421 static_cast<int64_t>(ntpFractionToMs(receive_f));
422 int64_t delayMs = static_cast<int64_t>(roundTripMs) - serverMs;
423 if (delayMs < 0) delayMs = 0;
424 if (delayMs > roundTripMs) delayMs = roundTripMs;
425 // Unix seconds, calculated with 32 bit (also if unsigned long has 64 bit):
426 // correct until 2106, also after the rollover of the NTP time in 2036
427 uint32_t transmitUnix = static_cast<uint32_t>(transmit - 2208988800UL);
428 uint64_t newTimeMs = static_cast<uint64_t>(transmitUnix) * 1000ULL +
429 ntpFractionToMs(transmit_f) +
430 static_cast<uint64_t>(delayMs / 2);
431 uint32_t newNtpTime = static_cast<uint32_t>(newTimeMs / 1000);
432 uint32_t newNtpTimeMs = static_cast<uint32_t>(newTimeMs % 1000);
433
434 if (!_hasBaseline) {
435 // First update: establish baseline for correction factor calculation
436 _baselineNtpTime = newNtpTime;
437 _baselineNtpTimeMs = newNtpTimeMs;
438 _baselineUpdateMillis = t3_millis;
439 _hasBaseline = true;
440 } else {
441 // Calculate correction factor based on cumulative drift since baseline
442 uint32_t raw_millis_elapsed = t3_millis - _baselineUpdateMillis;
443 // Calculate actual time elapsed according to NTP (with millisecond
444 // precision)
445 uint64_t actual_time_elapsed_ms =
446 ((uint64_t)(newNtpTime - _baselineNtpTime) * 1000ULL) +
447 newNtpTimeMs - _baselineNtpTimeMs;
448
449 // Only update correction factor if enough time has elapsed (at least 10
450 // seconds). This avoids division by zero and ensures meaningful
451 // calibration
452 if (raw_millis_elapsed >= 10000 && actual_time_elapsed_ms >= 10000) {
453 _millisCorrectionFactor = static_cast<double>(actual_time_elapsed_ms) /
454 static_cast<double>(raw_millis_elapsed);
455 }
456 }
457
458 _lastNtpTime = newNtpTime;
459 _lastNtpTimeMs = newNtpTimeMs;
460 _lastUpdateMillis = t3_millis;
461 _hasTime = true;
462 return true;
463 }

481 bool updateWithRTC() {
482 // Use system clock for transmit timestamp (uncorrected to avoid circular dependency)
483 uint32_t t0_unix = getUncorrectedTimeSec(); // seconds since 1970
484 uint32_t t0_ntp = t0_unix + 2208988800UL; // seconds since 1900
485 return ntpExchange(t0_ntp);
486 }
