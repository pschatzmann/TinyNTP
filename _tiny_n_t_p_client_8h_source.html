83 uint64_t getTimeMs() override {
84 if (_lastUpdateMillis == 0) {
85 return 0; // Time not yet initialized
86 }
87 // Calculate ms since last update with correction factor applied
88 uint32_t raw_ms_since_update = ::millis() - _lastUpdateMillis;
89 uint64_t corrected_ms_since_update = static_cast<uint64_t>(
90 static_cast<double>(raw_ms_since_update) * _millisCorrectionFactor);
91 // Return last NTP time in ms plus corrected elapsed ms, plus offset
92 return (static_cast<uint64_t>(_lastNtpTime) * 1000ULL) +
93 _lastNtpTimeMs +
94 corrected_ms_since_update +

316 bool ntpExchange(uint32_t txTm_s) {
317 NTPPacket packet = {};
318 // Set LI=3 (no warning), VN=3 (NTPv3), Mode=3 (client)
319 // Binary: 11 011 011 = 0xDB
320 packet.li_vn_mode = 0xDB;
321 // Convert transmit timestamp to network byte order
322 packet.txTm_s = l_htonl(txTm_s);
323 packet.txTm_f = 0;
324
325 _udp.begin(_localPort != 0 ? _localPort : _port);
326 _udp.beginPacket(_server, _port);
327 _udp.write(reinterpret_cast<uint8_t*>(&packet), sizeof(NTPPacket));
328 _udp.endPacket();
329
330 // Wait for response with timeout
331 uint32_t timeout_ms = ::millis() + _timeoutMs;
332 int packetSize = 0;
333 while ((packetSize = _udp.parsePacket()) == 0) {
334 if (::millis() > timeout_ms) {
335 log("NTP: request timed out\n");
336 return false;
337 }
338 }
339
340 // Read response
341 uint32_t t3_millis = ::millis();
342 NTPPacket response = {};
343 uint8_t* buffer = reinterpret_cast<uint8_t*>(&response);
344
345 // Check if packet size is correct
346 if (packetSize < (int)sizeof(NTPPacket)) {
347 _udp.read(buffer, packetSize); // Read available data for logging
348 log("NTP: packet too short - Expected: %d, Got: %d (%s)\n",
349 sizeof(NTPPacket), packetSize, buffer);
350 return false;
351 }
352
353 // Read the packet data - read in chunks if necessary
354 int totalRead = 0;
355 int bytesToRead = sizeof(NTPPacket);
356
357 while (totalRead < bytesToRead && _udp.available()) {
358 int bytesRead = _udp.read(buffer + totalRead, bytesToRead - totalRead);
359 if (bytesRead > 0) {
360 totalRead += bytesRead;
361 } else {
362 break;
363 }
364 }
365
366 if (totalRead < (int)sizeof(NTPPacket)) {
367 log("NTP: response read incomplete - Expected: %d, Got: %d (%s)\n",
368 sizeof(NTPPacket), totalRead, buffer);
369 return false;
370 }
371
372 bool useOffset = (txTm_s != 0);
373 if (useOffset) {
374 // Full NTP offset calculation
375 uint32_t originate = l_ntohl(response.origTm_s);
376 uint32_t receive = l_ntohl(response.rxTm_s);
377 uint32_t transmit = l_ntohl(response.txTm_s);
378 uint32_t transmit_f = l_ntohl(response.txTm_f);
379 // log("NTP: originate=%u, receive=%u, transmit=%u\n", originate, receive,
380 // transmit);
381 uint32_t t3_unix = getUncorrectedTimeSec();
382 uint32_t t3_ntp = t3_unix + 2208988800UL;
383 long offset =
384 ((long)(receive - originate) + (long)(transmit - t3_ntp)) / 2;
385 uint32_t newNtpTime = transmit - 2208988800UL + offset;
386 uint32_t newNtpTimeMs = ntpFractionToMs(transmit_f);
387
388 // Establish baseline on first update, calculate correction factor on subsequent updates
389 if (_baselineUpdateMillis == 0) {
390 // First update: establish baseline for correction factor calculation
391 _baselineNtpTime = newNtpTime;
392 _baselineNtpTimeMs = newNtpTimeMs;
393 _baselineUpdateMillis = t3_millis;
394 } else {
395 // Calculate correction factor based on cumulative drift since baseline
396 uint32_t raw_millis_elapsed = t3_millis - _baselineUpdateMillis;
397 // Calculate actual time elapsed according to NTP (with millisecond precision)
398 uint64_t actual_time_elapsed_ms =
399 ((uint64_t)(newNtpTime - _baselineNtpTime) * 1000ULL) +
400 newNtpTimeMs - _baselineNtpTimeMs;
401
402 // Only update correction factor if enough time has elapsed (at least 10 seconds)
403 // This avoids division by zero and ensures meaningful calibration
404 if (raw_millis_elapsed >= 10000 && actual_time_elapsed_ms >= 10000) {
405 _millisCorrectionFactor = static_cast<double>(actual_time_elapsed_ms) /
406 static_cast<double>(raw_millis_elapsed);
407 // log("NTP: correction factor = %.6f (actual=%llu ms, millis=%u ms from baseline)\n",
408 // _millisCorrectionFactor, actual_time_elapsed_ms, raw_millis_elapsed);
409 } else {
410 // log("NTP: skipping correction factor update - insufficient time from baseline (actual=%llu ms, millis=%u ms)\n",
411 // actual_time_elapsed_ms, raw_millis_elapsed);
412 }
413 }
414
415 _lastNtpTime = newNtpTime;
416 _lastNtpTimeMs = newNtpTimeMs;
417 _lastUpdateMillis = t3_millis;
418 } else {
419 // Only use server transmit timestamp (first update without RTC)
420 uint32_t transmit = l_ntohl(response.txTm_s);
421 uint32_t transmit_f = l_ntohl(response.txTm_f);
422 _lastNtpTime = transmit - 2208988800UL;
423 _lastNtpTimeMs = ntpFractionToMs(transmit_f);
424 _lastUpdateMillis = t3_millis;
425
426 // Establish baseline if not yet set
427 if (_baselineUpdateMillis == 0) {
428 _baselineNtpTime = _lastNtpTime;
429 _baselineNtpTimeMs = _lastNtpTimeMs;
430 _baselineUpdateMillis = t3_millis;
431 }
432 }
433 return true;
434 }

441 bool updateWithoutRTC() {
442 // No real time clock: set transmit timestamp to zero, use millis for
443 // timeout
444 return ntpExchange(0);
445 }
446
452 bool updateWithRTC() {
453 // Use system clock for transmit timestamp (uncorrected to avoid circular dependency)
454 uint32_t t0_unix = getUncorrectedTimeSec(); // seconds since 1970
455 uint32_t t0_ntp = t0_unix + 2208988800UL; // seconds since 1900
456 return ntpExchange(t0_ntp);
457 }
