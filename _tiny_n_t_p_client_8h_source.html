80 uint64_t getTimeMs() override {
81 if (_lastUpdateMillis == 0) {
82 return 0; // Time not yet initialized
83 }
84 // Calculate ms since last update with correction factor applied
85 uint32_t raw_ms_since_update = ::millis() - _lastUpdateMillis;
86 uint64_t corrected_ms_since_update = static_cast<uint64_t>(
87 static_cast<double>(raw_ms_since_update) * _millisCorrectionFactor);
88 // Return last NTP time in ms plus corrected elapsed ms, plus offset
89 return (static_cast<uint64_t>(_lastNtpTime) * 1000ULL) +
90 _lastNtpTimeMs +
91 corrected_ms_since_update +
92 static_cast<uint64_t>(_timeOffsetSeconds) * 1000ULL;
93 }

206 struct __attribute__((packed)) NTPPacket {
207 // Leap Indicator, Version Number, Mode (packed in one byte)
208 // Bit layout: LI(7-6) | VN(5-3) | Mode(2-0)
209 uint8_t li_vn_mode = 0; // Default: 0x00, set to 0xE3 for client request
210 uint8_t stratum = 0; // Stratum level of the local clock
211 uint8_t poll = 0; // Maximum interval between messages
212 uint8_t precision = 0; // Precision of the local clock
213 uint32_t rootDelay =
214 0; // Total round trip delay to the primary reference source
215 uint32_t rootDispersion =
216 0; // Max error relative to the primary reference source
217 uint32_t refId = 0; // Reference clock identifier
218 uint32_t refTm_s = 0; // Reference timestamp (seconds)
219 uint32_t refTm_f = 0; // Reference timestamp (fraction)
220 uint32_t origTm_s =
221 0; // Originate timestamp (seconds) - time request sent by client
222 uint32_t origTm_f = 0; // Originate timestamp (fraction)
223 uint32_t rxTm_s =
224 0; // Receive timestamp (seconds) - time request received by server

226 uint32_t txTm_s =
227 0; // Transmit timestamp (seconds) - time reply sent by server
228 uint32_t txTm_f = 0; // Transmit timestamp (fraction)

311 bool ntpExchange(uint32_t txTm_s) {
312 NTPPacket packet = {};
313 // Set LI=3 (no warning), VN=3 (NTPv3), Mode=3 (client)
314 // Binary: 11 011 011 = 0xDB
315 packet.li_vn_mode = 0xDB;
316 // Convert transmit timestamp to network byte order
317 packet.txTm_s = l_htonl(txTm_s);
318 packet.txTm_f = 0;
319
320 _udp.begin(_localPort != 0 ? _localPort : _port);
321 _udp.beginPacket(_server, _port);
322 _udp.write(reinterpret_cast<uint8_t*>(&packet), sizeof(NTPPacket));
323 _udp.endPacket();
324
325 // Wait for response with timeout
326 uint32_t timeout_ms = ::millis() + _timeoutMs;
327 int packetSize = 0;
328 while ((packetSize = _udp.parsePacket()) == 0) {
329 if (::millis() > timeout_ms) {
330 log("NTP: request timed out\n");
331 return false;
332 }
333 }
334
335 // Read response
336 uint32_t t3_millis = ::millis();
337 NTPPacket response = {};
338 uint8_t* buffer = reinterpret_cast<uint8_t*>(&response);
339
340 // Check if packet size is correct
341 if (packetSize < (int)sizeof(NTPPacket)) {
342 _udp.read(buffer, packetSize); // Read available data for logging
343 log("NTP: packet too short - Expected: %d, Got: %d (%s)\n",
344 sizeof(NTPPacket), packetSize, buffer);
345 return false;
346 }
347
348 // Read the packet data - read in chunks if necessary
349 int totalRead = 0;
350 int bytesToRead = sizeof(NTPPacket);
351
352 while (totalRead < bytesToRead && _udp.available()) {
353 int bytesRead = _udp.read(buffer + totalRead, bytesToRead - totalRead);
354 if (bytesRead > 0) {
355 totalRead += bytesRead;
356 } else {
357 break;
358 }
359 }
360
361 if (totalRead < (int)sizeof(NTPPacket)) {
362 log("NTP: response read incomplete - Expected: %d, Got: %d (%s)\n",
363 sizeof(NTPPacket), totalRead, buffer);
364 return false;
365 }
366
367 bool useOffset = (txTm_s != 0);
368 if (useOffset) {
369 // Full NTP offset calculation
370 uint32_t originate = l_ntohl(response.origTm_s);
371 uint32_t receive = l_ntohl(response.rxTm_s);
372 uint32_t transmit = l_ntohl(response.txTm_s);
373 uint32_t transmit_f = l_ntohl(response.txTm_f);
374 // log("NTP: originate=%u, receive=%u, transmit=%u\n", originate, receive,
375 // transmit);
376 uint32_t t3_unix = getUncorrectedTimeSec();
377 uint32_t t3_ntp = t3_unix + 2208988800UL;
378 long offset =
379 ((long)(receive - originate) + (long)(transmit - t3_ntp)) / 2;
380 uint32_t newNtpTime = transmit - 2208988800UL + offset;
381 uint32_t newNtpTimeMs = ntpFractionToMs(transmit_f);
382
383 // Establish baseline on first update, calculate correction factor on subsequent updates
384 if (_baselineUpdateMillis == 0) {
385 // First update: establish baseline for correction factor calculation
386 _baselineNtpTime = newNtpTime;
387 _baselineNtpTimeMs = newNtpTimeMs;
388 _baselineUpdateMillis = t3_millis;
389 } else {
390 // Calculate correction factor based on cumulative drift since baseline
391 uint32_t raw_millis_elapsed = t3_millis - _baselineUpdateMillis;
392 // Calculate actual time elapsed according to NTP (with millisecond precision)
393 uint64_t actual_time_elapsed_ms =
394 ((uint64_t)(newNtpTime - _baselineNtpTime) * 1000ULL) +
395 newNtpTimeMs - _baselineNtpTimeMs;
396
397 // Only update correction factor if enough time has elapsed (at least 10 seconds)
398 // This avoids division by zero and ensures meaningful calibration
399 if (raw_millis_elapsed >= 10000 && actual_time_elapsed_ms >= 10000) {
400 _millisCorrectionFactor = static_cast<double>(actual_time_elapsed_ms) /
401 static_cast<double>(raw_millis_elapsed);
402 // log("NTP: correction factor = %.6f (actual=%llu ms, millis=%u ms from baseline)\n",
403 // _millisCorrectionFactor, actual_time_elapsed_ms, raw_millis_elapsed);
404 } else {
405 // log("NTP: skipping correction factor update - insufficient time from baseline (actual=%llu ms, millis=%u ms)\n",
406 // actual_time_elapsed_ms, raw_millis_elapsed);
407 }
408 }
409
410 _lastNtpTime = newNtpTime;
411 _lastNtpTimeMs = newNtpTimeMs;
412 _lastUpdateMillis = t3_millis;
413 } else {
414 // Only use server transmit timestamp (first update without RTC)
415 uint32_t transmit = l_ntohl(response.txTm_s);
416 uint32_t transmit_f = l_ntohl(response.txTm_f);
417 _lastNtpTime = transmit - 2208988800UL;
418 _lastNtpTimeMs = ntpFractionToMs(transmit_f);
419 _lastUpdateMillis = t3_millis;
420
421 // Establish baseline if not yet set
422 if (_baselineUpdateMillis == 0) {
423 _baselineNtpTime = _lastNtpTime;
424 _baselineNtpTimeMs = _lastNtpTimeMs;
425 _baselineUpdateMillis = t3_millis;
426 }
427 }
428 return true;
429 }

436 bool updateWithoutRTC() {
437 // No real time clock: set transmit timestamp to zero, use millis for
438 // timeout
439 return ntpExchange(0);
440 }
441
447 bool updateWithRTC() {
448 // Use system clock for transmit timestamp (uncorrected to avoid circular dependency)
449 uint32_t t0_unix = getUncorrectedTimeSec(); // seconds since 1970
450 uint32_t t0_ntp = t0_unix + 2208988800UL; // seconds since 1900
451 return ntpExchange(t0_ntp);
452 }
