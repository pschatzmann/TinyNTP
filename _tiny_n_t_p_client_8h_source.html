104 uint64_t getTimeMs() override {
105 if (!_hasTime) {
106 return 0; // Time not yet initialized
107 }
108 // Calculate ms since last update with correction factor applied
109 uint32_t raw_ms_since_update = ::millis() - _lastUpdateMillis;
110 uint64_t corrected_ms_since_update = static_cast<uint64_t>(
111 static_cast<double>(raw_ms_since_update) * _millisCorrectionFactor);
112 // Return last NTP time in ms plus corrected elapsed ms, plus offset
113 return (static_cast<uint64_t>(_lastNtpTime) * 1000ULL) +
114 _lastNtpTimeMs +
115 corrected_ms_since_update +
116 static_cast<uint64_t>(_timeOffsetSeconds) * 1000ULL;
117 }

358 bool readResponse(int packetSize, NTPPacket& response) {
359 uint8_t* buffer = reinterpret_cast<uint8_t*>(&response);
360 int totalRead = 0;
361 while (totalRead < (int)sizeof(NTPPacket) && _udp.available()) {
362 int bytesRead =
363 _udp.read(buffer + totalRead, (int)sizeof(NTPPacket) - totalRead);
364 if (bytesRead <= 0) break;
365 totalRead += bytesRead;
366 }
367 uint8_t rest[16];
368 while (_udp.available() > 0 && _udp.read(rest, sizeof(rest)) > 0) {
369 }
370 if (packetSize < (int)sizeof(NTPPacket)) {
371 log("NTP: packet too short - Expected: %d, Got: %d\n",
372 (int)sizeof(NTPPacket), packetSize);
373 return false;
374 }
375 if (totalRead < (int)sizeof(NTPPacket)) {
376 log("NTP: response read incomplete - Expected: %d, Got: %d\n",
377 (int)sizeof(NTPPacket), totalRead);
378 return false;
379 }
380 return true;
381 }

392 bool ntpExchange(uint32_t txTm_s) {
393 NTPPacket packet = {};
394 // Set LI=3 (clock not synchronized), VN=3 (NTPv3), Mode=3 (client)
395 // Binary: 11 011 011 = 0xDB
396 packet.li_vn_mode = 0xDB;
397 // Convert transmit timestamp to network byte order
398 packet.txTm_s = l_htonl(txTm_s);
399
400 // The socket stays open between the updates, so that a late response is
401 // still received (and ignored) and does not get lost with the socket
402 uint16_t localPort = _localPort != 0 ? _localPort : _port;
403 if (_openPort != localPort) {
404 _openPort = _udp.begin(localPort) ? localPort : 0;
405 }
406
407 // The request is sent again until a response arrives: some WiFi networks
408 // (e.g. with an ESP32) lose responses or hold them back until the device
409 // sends the next packet. Each request has its own number in the transmit
410 // fraction, which the server returns as originate time: the round trip is
411 // measured from the request that was answered, and responses to the
412 // requests of an earlier update are ignored.
413 uint32_t sentMillis[MaxRequests];
414 int sentCount = 0;
415 uint32_t requestId = ++_requestCounter << 8;
416 uint32_t startMillis = ::millis();
417 uint32_t t0_millis = 0; // the time when the answered request was sent
418 uint32_t t3_millis = 0; // the time when the response was received
419 NTPPacket response = {};
420 bool received = false;
421 while (!received) {
422 // 32 bit arithmetic: also correct after the overflow of millis(), and if
423 // millis() returns a 64 bit value (e.g. on a computer)
424 uint32_t now = ::millis();
425 if (static_cast<uint32_t>(now - startMillis) > _timeoutMs) {
426 log("NTP: request timed out\n");
427 return false;
428 }
429 if (sentCount < MaxRequests &&
430 (sentCount == 0 || static_cast<uint32_t>(
431 now - sentMillis[sentCount - 1]) >=
432 ResendIntervalMs)) {
433 packet.txTm_f = l_htonl(requestId | sentCount);
434 sentMillis[sentCount++] = now;
435 _udp.beginPacket(_server, _port);
436 _udp.write(reinterpret_cast<uint8_t*>(&packet), sizeof(NTPPacket));
437 _udp.endPacket();
438 }
439
440 int packetSize = _udp.parsePacket();
441 if (packetSize <= 0) continue;
442 t3_millis = ::millis();
443 if (!readResponse(packetSize, response)) continue;
444 uint32_t originate = l_ntohl(response.origTm_f);
445 int index = static_cast<int>(originate & 0xFF);
446 if ((originate & ~0xFFu) != requestId || index >= sentCount) continue;
447 t0_millis = sentMillis[index];
448 received = true;
449 }
450
451 uint32_t receive = l_ntohl(response.rxTm_s);
452 uint32_t receive_f = l_ntohl(response.rxTm_f);
453 uint32_t transmit = l_ntohl(response.txTm_s);
454 uint32_t transmit_f = l_ntohl(response.txTm_f);
455
456 // The transmit time of the server is the time when it sent the response.
457 // The response needed about half of the network delay to get to us, which
458 // is the time of the round trip without the time that the server needed.
459 uint32_t roundTripMs = t3_millis - t0_millis;
460 int64_t serverMs =
461 static_cast<int64_t>(static_cast<int32_t>(transmit - receive)) * 1000 +
462 static_cast<int64_t>(ntpFractionToMs(transmit_f)) -
463 static_cast<int64_t>(ntpFractionToMs(receive_f));
464 int64_t delayMs = static_cast<int64_t>(roundTripMs) - serverMs;
465 if (delayMs < 0) delayMs = 0;
466 if (delayMs > roundTripMs) delayMs = roundTripMs;
467 // Unix seconds, calculated with 32 bit (also if unsigned long has 64 bit):
468 // correct until 2106, also after the rollover of the NTP time in 2036
469 uint32_t transmitUnix = static_cast<uint32_t>(transmit - 2208988800UL);
470 uint64_t newTimeMs = static_cast<uint64_t>(transmitUnix) * 1000ULL +
471 ntpFractionToMs(transmit_f) +
472 static_cast<uint64_t>(delayMs / 2);
473 uint32_t newNtpTime = static_cast<uint32_t>(newTimeMs / 1000);
474 uint32_t newNtpTimeMs = static_cast<uint32_t>(newTimeMs % 1000);
475
476 if (!_hasBaseline) {
477 // First update: establish baseline for correction factor calculation
478 _baselineNtpTime = newNtpTime;
479 _baselineNtpTimeMs = newNtpTimeMs;
480 _baselineUpdateMillis = t3_millis;
481 _hasBaseline = true;
482 } else {
483 // Calculate correction factor based on cumulative drift since baseline
484 uint32_t raw_millis_elapsed = t3_millis - _baselineUpdateMillis;
485 // Calculate actual time elapsed according to NTP (with millisecond
486 // precision)
487 uint64_t actual_time_elapsed_ms =
488 ((uint64_t)(newNtpTime - _baselineNtpTime) * 1000ULL) +
489 newNtpTimeMs - _baselineNtpTimeMs;
490
491 // Only update correction factor if enough time has elapsed (at least 10
492 // seconds). This avoids division by zero and ensures meaningful
493 // calibration
494 if (raw_millis_elapsed >= 10000 && actual_time_elapsed_ms >= 10000) {
495 _millisCorrectionFactor = static_cast<double>(actual_time_elapsed_ms) /
496 static_cast<double>(raw_millis_elapsed);
497 }
498 }
499
500 _lastNtpTime = newNtpTime;
501 _lastNtpTimeMs = newNtpTimeMs;
502 _lastUpdateMillis = t3_millis;
503 _hasTime = true;
504 return true;
505 }

512 bool updateWithoutRTC() {
513 // No real time clock: set transmit timestamp to zero, use millis for
514 // timeout
515 return ntpExchange(0);
516 }
517
523 bool updateWithRTC() {
524 // Use system clock for transmit timestamp (uncorrected to avoid circular dependency)
525 uint32_t t0_unix = getUncorrectedTimeSec(); // seconds since 1970
526 uint32_t t0_ntp = t0_unix + 2208988800UL; // seconds since 1900
527 return ntpExchange(t0_ntp);
528 }
