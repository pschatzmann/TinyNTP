104 static void makeNonce(const uint8_t* key, size_t keyLen, uint32_t counter,
105 uint64_t timeMs, uint32_t millis, uint8_t* nonce) {
106 uint8_t data[16];
107 put32(data, counter);
108 put64(data + 4, timeMs);
109 put32(data + 12, millis);
110 uint8_t mac[TinySHA256::DigestSize];
111 TinyHMACSHA256::calculate(key, keyLen, data, sizeof data, mac);
112 memcpy(nonce, mac, NonceSize);
113 }

116 static size_t makeSetRequest(uint8_t* out, const uint8_t* key, size_t keyLen,
117 const uint8_t* nonce, uint64_t timeMs) {
118 out[0] = Header;
119 out[1] = SetRequest;
120 memcpy(out + 2, nonce, NonceSize);
121 put64(out + 2 + NonceSize, timeMs);
122 // the HMAC covers the type, the nonce and the time
123 TinyHMACSHA256::calculate(key, keyLen, out + 1, 1 + NonceSize + 8,
124 out + 2 + NonceSize + 8);
125 return SetRequestSize;
126 }

139 static bool verifySetRequest(const uint8_t* msg, size_t len,
140 const uint8_t* key, size_t keyLen,
141 const uint8_t* nonce, uint64_t& timeMs) {
142 if (len < SetRequestSize || !isPrivate(msg, len) || msg[1] != SetRequest) {
143 return false;
144 }
145 uint8_t mac[TinySHA256::DigestSize];
146 TinyHMACSHA256::calculate(key, keyLen, msg + 1, 1 + NonceSize + 8, mac);
147 bool ok = TinyHMACSHA256::equals(mac, msg + 2 + NonceSize + 8,
148 TinySHA256::DigestSize);
149 ok = TinyHMACSHA256::equals(nonce, msg + 2, NonceSize) && ok;
150 if (ok) timeMs = get64(msg + 2 + NonceSize);
151 return ok;
152 }
153
155 static size_t makeSetResponse(uint8_t* out, Status status, uint64_t timeMs) {
156 out[0] = Header;
157 out[1] = SetResponse;
158 out[2] = status;
159 put64(out + 3, timeMs);
160 return SetResponseSize;
161 }
162
164 static bool parseSetResponse(const uint8_t* msg, size_t len, Status& status,
165 uint64_t& timeMs) {
166 if (len < SetResponseSize || !isPrivate(msg, len) ||
167 msg[1] != SetResponse) {
168 return false;
169 }
170 status = static_cast<Status>(msg[2]);
171 timeMs = get64(msg + 3);
172 return true;
173 }

175 protected:
177 static void put32(uint8_t* out, uint32_t v) {
178 for (int i = 0; i < 4; i++) out[i] = v >> (24 - 8 * i);
179 }
180
181 static void put64(uint8_t* out, uint64_t v) {
182 for (int i = 0; i < 8; i++) out[i] = v >> (56 - 8 * i);
183 }

185 static uint64_t get64(const uint8_t* in) {
186 uint64_t v = 0;
187 for (int i = 0; i < 8; i++) v = (v << 8) | in[i];
188 return v;
189 }
